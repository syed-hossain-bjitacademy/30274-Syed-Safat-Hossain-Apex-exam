    for (Opportunity opp : Trigger.new) {
        Account acc = [SELECT Id, Strategic__c, (SELECT Id, Amount FROM Opportunities) FROM Account WHERE Id = :opp.AccountId];
        List<Opportunity> opportunities = [SELECT Id, Amount FROM Opportunity WHERE AccountId = :opp.AccountId];
        
        Decimal totalAmount = 0;
        for (Account item : acc) {
            if (item.Opportunity.Amount != null) {
                totalAmount += item.Opportunity.Amount;
            }
        }
        if (totalAmount >= 5000000) {
            acc.Strategic__c = true;
            update acc;
        }
    }
